# Face Recognition Pipeline — Dependencies
# ─────────────────────────────────────────

# ── Core Libraries ──
opencv-python>=4.8.0     # For camera capture, image processing, and display
numpy>=1.24.0            # For array manipulation and embedding math

# ── Face Detection & Recognition (InsightFace) ──
# onnxruntime-gpu MUST stay on a CUDA 12 build: 1.27+ links libcudart.so.13
# (CUDA 13) and fails to import against the cu12 wheels pinned below.
# 1.22.0 is verified working on the production L4 VPS.
onnxruntime-gpu>=1.16.0,<1.23   # For NVIDIA GPUs (CUDA 12 + cuDNN 9 builds)
insightface>=0.7.3       # For production detection and embedding (ArcFace/YOLO)
# NOTE: insightface declares a dependency on the CPU 'onnxruntime' package;
# pip installs it alongside onnxruntime-gpu and the CPU wheel clobbers the
# shared module files, silently downgrading inference to CPU. After installing,
# uninstall the CPU wheel and force-reinstall onnxruntime-gpu with `--no-deps`.

# ── CUDA 12 Runtime (pip wheels, for onnxruntime-gpu) ──
# These provide the CUDA/cuDNN libraries onnxruntime-gpu needs at runtime,
# so only the NVIDIA driver is required on the host — no system CUDA toolkit.
# Windows: engine/pipeline/embedding.py registers the wheel DLL dirs.
# Linux: add the pip wheel library directories to `LD_LIBRARY_PATH`.
# Not needed if you have CUDA Toolkit 12.x + cuDNN 9 installed system-wide.
nvidia-cublas-cu12>=12.4
nvidia-cuda-runtime-cu12>=12.4
nvidia-cuda-nvrtc-cu12>=12.4
nvidia-cudnn-cu12>=9.0
nvidia-cufft-cu12>=11.0
nvidia-curand-cu12>=10.3
nvidia-cusparse-cu12>=12.3
nvidia-cusolver-cu12>=11.6
